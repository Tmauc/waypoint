## Step 1 — Define your schema

A `WaypointSchema` describes every step, field, validation rule and visibility condition in your journey.

```ts
// lib/onboarding.schema.ts
import type { WaypointSchema } from "@waypointjs/core";

export const onboardingSchema: WaypointSchema = {
  version: "1",
  id: "onboarding",
  name: "User Onboarding",
  steps: [
    {
      id: "personal",
      title: "Personal info",
      url: "/onboarding/personal",
      fields: [
        {
          id: "name",
          type: "text",
          label: "Full name",
          validation: [{ type: "required", message: "Name is required" }],
        },
        {
          id: "email",
          type: "email",
          label: "Email address",
          validation: [
            { type: "required", message: "Email is required" },
            { type: "email",    message: "Must be a valid email" },
          ],
        },
      ],
    },
    {
      id: "company",
      title: "Company",
      url: "/onboarding/company",
      // This step only appears when the user selects "professional"
      visibleWhen: {
        combinator: "and",
        rules: [{ field: "personal.accountType", operator: "equals", value: "professional" }],
      },
      fields: [
        { id: "companyName", type: "text", label: "Company name",
          validation: [{ type: "required", message: "Required" }] },
      ],
    },
    {
      id: "confirm",
      title: "Review & confirm",
      url: "/onboarding/confirm",
      fields: [],
    },
  ],
};
```

<Callout type="tip">
  You can build schemas visually with `<WaypointBuilder>` instead of writing JSON by hand. See the [Builder guide](/guides/builder).
</Callout>

## Step 2 — Add WaypointRunner to your layout

`WaypointRunner` is a React provider that creates an isolated Zustand store for your journey and handles init, persistence and callbacks.

```tsx
// app/onboarding/layout.tsx
"use client";

import { WaypointRunner } from "@waypointjs/next";
import { onboardingSchema } from "@/lib/onboarding.schema";

export default function OnboardingLayout({ children }: { children: React.ReactNode }) {
  async function handleComplete(data: Record<string, Record<string, unknown>>) {
    await fetch("/api/onboarding", { method: "POST", body: JSON.stringify(data) });
    router.push("/dashboard");
  }

  return (
    <WaypointRunner schema={onboardingSchema} onComplete={handleComplete}>
      {children}
    </WaypointRunner>
  );
}
```

## Step 3 — Use `useWaypointStep` in each page

`useWaypointStep()` auto-detects the current step from the URL, returns the visible fields, and provides a react-hook-form instance pre-wired to the step's Zod validation schema.

```tsx
// app/onboarding/personal/page.tsx
"use client";

import { useWaypointStep } from "@waypointjs/next";

export default function PersonalPage() {
  const { fields, form, handleSubmit, goBack, progress, isFirstStep, isLastStep } =
    useWaypointStep();

  return (
    <div>
      {/* Progress bar */}
      <div style={{ width: `${progress}%`, height: 4, background: "blue" }} />

      <form onSubmit={form.handleSubmit(handleSubmit)}>
        {fields.map((field) => (
          <div key={field.definition.id}>
            <label>{field.definition.label}</label>
            <input
              type={field.definition.type}
              placeholder={field.definition.placeholder}
              {...form.register(field.definition.id)}
            />
            {form.formState.errors[field.definition.id] && (
              <span>{String(form.formState.errors[field.definition.id]?.message)}</span>
            )}
          </div>
        ))}

        {!isFirstStep && (
          <button type="button" onClick={goBack}>← Back</button>
        )}
        <button type="submit">
          {isLastStep ? "Finish ✓" : "Continue →"}
        </button>
      </form>
    </div>
  );
}
```

<Callout>
  The same component works for **every step** — fields are resolved dynamically from the schema. You only need one page component per step if your fields are similar enough.
</Callout>

## Step 4 — Set up routes

Each step's `url` in your schema must match a Next.js route. For the schema above:

```
app/onboarding/
├── layout.tsx          ← WaypointRunner lives here
├── personal/
│   └── page.tsx        ← step id: "personal"
├── company/
│   └── page.tsx        ← step id: "company" (conditional)
└── confirm/
    └── page.tsx        ← step id: "confirm"
```

`useWaypointStep()` matches the current `pathname` to the step `url`, so each page automatically knows which step it is.
